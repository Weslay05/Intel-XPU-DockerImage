#! /run/current-system/sw/bin/bash

# For Device Drivers
--device /dev/dri
--privileged

# For inter process memory communication
-shm 4g
-ipc host

--network ["bridge", "host", "..."]

-v path-or-volume:mount-point
-w working-dir
-p port-host:port-docker  # docker port has to be exposed to all interfaces (0.0.0.0)
--name container-name  # for easier managment
-e env-vari=vari

# conda one liner
	conda run --no-capture-output -p conda \
	python main.py --listen 0.0.0.0;
